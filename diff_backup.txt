+/// A specialized menu button for selecting multiple languages
+struct LiquidGlassMultiLanguageMenu: View {
+    let items: [LanguageOption]
+    @Binding var selection: String

+    var body: some View {

+                if let children = lang.children {
+                    Menu(lang.name) {
+                        ForEach(children, id: \.id) { child in
+                            Button {
+                                toggleSelection(child.id)
+                            } label: {
+                                if selectedIds.contains(child.id) {
+                                    Label(child.name, systemImage: "checkmark")
+                                } else {
+                                    Text(child.name)
+                                }
+                            }
+                        }
+                    }
+                } else {
+                    Button {
+                        toggleSelection(lang.id)
+                    } label: {
+                        if selectedIds.contains(lang.id) {
+                            Label(lang.name, systemImage: "checkmark")
+                        } else {
+                            Text(lang.name)
+                        }
+                    }
+                }
+            }
+        } label: {
+            HStack(spacing: 6) {
+                let name = selectedIds.isEmpty ? "All Languages" : "\(selectedIds.count) Selected"
+                
+                Text(name)
+                    .font(.system(size: 12, weight: .medium, design: .rounded))
+                    .foregroundStyle(.primary)
+                    .lineLimit(1)
+
+                Spacer(minLength: 4)
+
+                Image(systemName: "chevron.up.chevron.down")
+                    .font(.system(size: 9, weight: .bold))
+                    .foregroundStyle(.secondary)
+            }

+            .padding(.vertical, 6)

+                        .fill(Color.primary.opacity(0.05))
+
+                    RoundedRectangle(cornerRadius: 10)
+                        .fill(.ultraThinMaterial.opacity(0.6))
+                }
+            )
+            .overlay(
+                RoundedRectangle(cornerRadius: 10)
+                    .strokeBorder(Color.primary.opacity(0.12), lineWidth: 0.5)
+            )
+        }
+        .buttonStyle(.plain)
+    }
+    
+    private func toggleSelection(_ id: String) {
+        var current = selectedIds
+        if current.contains(id) {
+            current.removeAll(where: { $0 == id })
+        } else {
+            if current.count < 3 {
+                current.append(id)
+            } else {
+                // Remove first, add new
+                current.removeFirst()
+                current.append(id)
+            }
+        }
+        selection = current.joined(separator: ",")
+    }
